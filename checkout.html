import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-app.js"; import { getFirestore, doc, getDoc, setDoc } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js"; // Aapki Firebase Configuration (Yahan apni details dalein) const firebaseConfig = { apiKey: "AQUAP_AP_YOUR_KEY", authDomain: "your-app.firebaseapp.com", projectId: "your-app-id", }; const app = initializeApp(firebaseConfig); const db = getFirestore(app); // URL se Invoice ID nikalna (e.g. checkout.html?invoice=INV101) const urlParams = new URLSearchParams(window.location.search); const invoiceId = urlParams.get('invoice') || "INV_TEST"; async function loadInvoiceDetails() { try { // Firestore se dukandaar aur bill ki details laana const docRef = doc(db, "invoices", invoiceId); const docSnap = await getDoc(docRef); if (docSnap.exists()) { const data = docSnap.data(); document.getElementById('shopName').innerText = data.shopName; document.getElementById('billAmount').innerText = "₹" + data.amount; document.getElementById('upiIdText').innerText = data.upiId; document.getElementById('qrCodeImg').src = `https://api.qrserver.com/v1/create-qr-code/?size=200x200&data=upi://pay?pa=${data.upiId}&pn=${data.shopName}&am=${data.amount}&cu=INR`; } else { alert("Invoice not found!"); } } catch (error) { console.error("Error loading invoice:", error); } } // Global scope mein function attach karna taaki HTML button click kaam kare window.submitUtr = async function() { const utr = document.getElementById('utrInput').value.trim(); if (utr.length < 12) { alert("Kripya poora 12-digit UTR number dalein."); return; } document.getElementById('submitBtn').innerText = "Submitting..."; document.getElementById('submitBtn').disabled = true; try { // UTR ko database mein save karna (Status: Pending) await setDoc(doc(db, "invoices", invoiceId), { utrNumber: utr, status: "Pending Verification", submittedAt: new Date().toISOString() }, { merge: true }); document.getElementById('statusMessage').innerHTML = "✓ UTR Submitted Successfully! Waiting for merchant approval."; document.querySelector('.bg-amber-50').style.display = 'none'; } catch (error) { console.error("Error submitting UTR:", error); alert("Kuch galat ho gaya. Phir se try karein."); document.getElementById('submitBtn').innerText = "Submit UTR & Verify"; document.getElementById('submitBtn').disabled = false; } } loadInvoiceDetails();
